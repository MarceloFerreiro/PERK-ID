\begin{frame}[fragile]{Descriptores visuales utilizados}
  \begin{columns}[T,onlytextwidth]
    \column{0.50\textwidth}
      \begin{block}{Configuración activa}
      \begin{lstlisting}
[features]
enabled = ["SIFT", "GLCM", "HU_MOMENTS", "LBP"]

[features.bow]
n_clusters = 250
      \end{lstlisting}
      \end{block}
    \column{0.46\textwidth}
      \begin{itemize}
        \item \textbf{SIFT}: puntos locales, logos, letras y bordes.
        \item \textbf{BoW}: convierte SIFT variable en vector fijo.
        \item \textbf{GLCM}: textura global de grises.
        \item \textbf{Hu Moments}: forma global.
        \item \textbf{LBP}: microtexturas y patrones locales.
      \end{itemize}
  \end{columns}

\end{frame}

\begin{frame}{Cómo se construye el índice}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \begin{enumerate}
        \item Leer cada imagen del dataset.
        \item Extraer GLCM, Hu Moments y LBP.
        \item Extraer descriptores SIFT.
        \item Entrenar KMeans para Bag of Words.
        \item Concatenar todo en un vector final.
        \item Guardar matriz de features e imágenes asociadas.
      \end{enumerate}
    \column{0.48\textwidth}
      \begin{tikzpicture}[node distance=0.45cm]

\begin{frame}[fragile]{Consulta y ranking}
  \begin{columns}[T,onlytextwidth]
    \column{0.52\textwidth}
      \begin{block}{\codefile{src/Ranker.py}}
      Recibe la imagen de consulta, calcula sus features y busca los vecinos más cercanos en el índice.
      \end{block}
      \begin{itemize}
        \item Métrica configurada: similitud por distancia coseno.
        \item Motor: \codefile{NearestNeighbors} de scikit-learn.
        \item Devuelve índices, distancias y rutas de las imágenes más parecidas.
      \end{itemize}
    \column{0.42\textwidth}
      \begin{lstlisting}
[ranker]
metric = 'cosine'
n_neighbors = 10
      \end{lstlisting}
      \vspace{0.3cm}
      \begin{block}{Interpretación}
      Menor distancia implica mayor similitud visual respecto a la foto de entrada.
      \end{block}
  \end{columns}

\end{frame}

\begin{frame}{Frontend Flutter}
  \begin{columns}[T,onlytextwidth]

      Implementa la interfaz móvil/web: cámara, galería, configuración y resultados.
